\begin{frame}{The +29\,\%: muons the 2018 weighting dropped}
  \begin{columns}[c]
    \begin{column}{0.52\textwidth}\footnotesize
      \texttt{extractMuonsAndUpdateWeight.py} divides the weight of \key{every} muon whose mother is $\eta,\omega,\phi,\rho^0,\eta'$ by 100, assuming its channel was boosted $\times100$.\\[1.5mm]
      But the boost hit only hard-coded channel numbers of Pythia 8.223:
      \begin{itemize}\footnotesize
        \item $\eta\to\mu\mu\gamma$ (ch.\,6, BR $3.1\cdot10^{-4}$): \key{not boosted} (product order fails the check)
        \item $\omega\to\pi^0\mu\mu$ (ch.\,5): \key{not boosted}
        \item $\eta$, $\eta'$ made in the cascade: Geant4 had no muon channels at all
      \end{itemize}
      $\Rightarrow$ these muons enter with $1/100$ of their weight, effectively \key{dropped}.\\[1.5mm]
      Seen in the 2018 files: $\eta$ muon pairs are \key{73\,\% pure $\mu\mu$} (physically 2\,\%).
    \end{column}
    \begin{column}{0.44\textwidth}\footnotesize
      \begin{tabular}{@{}lr@{}}
        \toprule
        ours / 2018 sample ($p>25$\,GeV) & ratio\\
        \midrule
        physical weights & \num{1.29}\\
        with the 2018 weighting & \num{1.016}\\
        per momentum bin, 25--400\,GeV & 0.99--1.03\\
        $p_T$ shape, 0--2.7\,GeV & 1.00\\
        \bottomrule
      \end{tabular}\\[3mm]
      \takeaway[\footnotesize]{The 2018 sample lacks \key{$\sim$22\,\%} of the physical muon flux, at \key{low $p_T$} (Dalitz decays).}\\[2mm]
      \aside{Low-$p_T$ muons are mostly swept by the shield; whether it changes the veto rates needs the shield transport.}

  \end{columns}
\end{frame}

\blurb{A generator that produces every muon source at its physical rate, with the rare ones forced and weighted.}
\section{A complete target generator}

\begin{frame}{Every source, forced and weighted}
  \begin{columns}[c]
    \begin{column}{0.56\textwidth}\footnotesize
      Geant4 cascade in the target (\texttt{tailtransport}, complete mode), with Pythia\,8 events as primaries:
      \begin{itemize}\footnotesize
        \item \key{long-lived} ($\pi$, K, hyperons): decay hazard per step, forced into the muon channels, weight = BR $\times$ survival; Geant4's kinematics where it has the channel
        \item \key{prompt} ($\eta,\omega,\rho,\phi,\eta'$, J/$\psi$, D, B, $\tau$, \dots): decayed by an embedded Pythia, natural + forced, wherever they appear (also deep in the cascade)
        \item \key{resonances inside hadronic models}: Geant4's tables replaced by Pythia's, muons tagged, $\times100$
        \item $\gamma\to\mu\mu$ $\times500$, $e^+e^-\to\mu\mu$ $\times100$; the Pythia event's own muons kept
        \item charm / beauty hadrons from the Pythia\,6 cascade, decayed the same way
      \end{itemize}
      \textbf{Analog mode}: the same code with nothing forced, the reference.
    \end{column}
    \begin{column}{0.4\textwidth}\footnotesize
      \key{A hole in Geant4}:\\[1mm]
      Geant4's decay tables have \key{no semileptonic muon channels} for K$_S$, $\Lambda$, $\Sigma^-$, $\Xi$ (BR $1.6$--$4.6\cdot10^{-4}$). Unforced Geant4 (and the 2018 sample) never makes these muons. We add them to Geant4's tables.\\[2mm]
      Small: K$_S$ muons are $\sim$0.4\,\% of the min.\ bias flux.\\[2mm]
      \takeaway[\footnotesize]{Forced and unforced use the same channel list: the forcing changes only the statistics, never the physics.}
    \end{column}
  \end{columns}
\end{frame}

        \item complete mode: every muon source
      \end{itemize}
      Made on CSCS eiger: 3\,000 min.\ bias slices on 24 nodes in 2\,h, plus the onia, charm and beauty runs.\\[1mm]
      \aside{Everything to reproduce it (commands, versions, seeds, code, checksums) is in the dataset's README.}
    \end{column}

\begin{frame}{Importance sampling: CPU where the veto needs it}
  \begin{columns}[c]
    \begin{column}{0.58\textwidth}
      \livewidget{rr_map}{\linewidth}{0.5\textheight}\\[0.5mm]
      \begin{columns}[c]
        \begin{column}{0.5\linewidth}\sld{rr_qmin}\\\sld{rr_e1}\end{column}

    \begin{column}{0.4\textwidth}\scriptsize

      \key{Russian roulette} on every particle (Pythia event and cascade): kept with
      \[ q = \min\Bigl(1,\ \max\bigl(q_\text{min}, (E/E_1)^N\bigr)\,\max\bigl(1, (p_T/p_{T0})^2\bigr)\Bigr), \]

      \key{Weight window} on the recorded muons: per $(p, p_T)$ cell, rows below $0.05\,\sum w^2/\sum w$ are rouletted too.\\[1mm]

        \toprule

        \quad $n_\text{eff}$, $\chi^2/\text{ndf}$ & $-9$\,\%, 48.7 / 40\\
        \bottomrule
      \end{tabular}\\[1mm]

  \end{columns}

\begin{frame}{A storage rule that knows where muons cross}
  \begin{columns}[c]
    \begin{column}{0.44\textwidth}\scriptsize
      Per seed, each muon of the common types is kept with
      \[ p_i = \max\Bigl(p_\text{floor},\ \min\bigl(1,\ \lambda\, w_i \max_{c \ni i} \tfrac{1}{V_c}\bigr)\Bigr) \]
      \vspace{-3mm}
      \begin{itemize}\scriptsize
        \item $c$: probe cells (25\,cm) on its path; $V_c$: muons per spill there, from the seed itself
        \item weight $w_i/p_i$ (Horvitz--Thompson): unbiased
        \item rare cells keep everything, busy cells a fraction
      \end{itemize}
      \vspace{1mm}
      Tested on 800 seeds: weakest cells (10th percentile) \key{23--80$\times$} better than the uniform rule over 6\,000 seeds, with fewer rows. Cells below 10 muons: 30--44\,\% $\to$ 0--1\,\%.\\[1mm]
