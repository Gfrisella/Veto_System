# tailtransport (cpp/tailgen/tailtransport.cc) against Geant4 10.01.p03 and Pythia 8.223: the 2018 FairShip production's
# versions (shipdist), for the complete / analog modes with the 2018 software. bin/g4_101.sh bash cpp/tailgen_g4101/build.sh
cmake_minimum_required(VERSION 3.16)
project(tailtransport_g4101 CXX)
set(CMAKE_CXX_STANDARD 14)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(PYBIND11_FINDPYTHON ON)
find_package(pybind11 CONFIG REQUIRED)
find_package(Geant4 REQUIRED)
pybind11_add_module(tailtransport ${CMAKE_CURRENT_SOURCE_DIR}/../tailgen/tailtransport.cc)
target_include_directories(tailtransport PRIVATE ${Geant4_INCLUDE_DIRS} ${PYTHIA8_223}/include)
target_compile_definitions(tailtransport PRIVATE TT_WITH_PYTHIA)
target_link_libraries(tailtransport PRIVATE ${Geant4_LIBRARIES} ${PYTHIA8_223}/lib/libpythia8.a dl)
set_target_properties(tailtransport PROPERTIES INSTALL_RPATH "${Geant4_DIR}/../.." BUILD_RPATH_USE_ORIGIN OFF)
