cmake_minimum_required(VERSION 3.16)
project(tailtransport CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(PYBIND11_FINDPYTHON ON)
find_package(pybind11 CONFIG REQUIRED)
find_package(Geant4 REQUIRED)
pybind11_add_module(tailtransport tailtransport.cc)
target_include_directories(tailtransport PRIVATE ${Geant4_INCLUDE_DIRS})
target_link_libraries(tailtransport PRIVATE ${Geant4_LIBRARIES})
# Pythia 8 (decayer of the complete / analog modes); without it only the legacy mode is built
find_library(PYTHIA8_LIB pythia8 HINTS /usr/local/lib)
find_path(PYTHIA8_INC Pythia8/Pythia.h HINTS /usr/local/include)
if(PYTHIA8_LIB AND PYTHIA8_INC)
  target_compile_definitions(tailtransport PRIVATE TT_WITH_PYTHIA)
  target_include_directories(tailtransport PRIVATE ${PYTHIA8_INC})
  target_link_libraries(tailtransport PRIVATE ${PYTHIA8_LIB})
  message(STATUS "tailtransport: with Pythia 8 (${PYTHIA8_LIB})")
endif()
